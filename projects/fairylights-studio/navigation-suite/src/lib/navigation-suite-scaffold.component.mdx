## Getting Started

### 1. Import the module

<Source
  language="ts"
  code={`import { MAT_NAVIGATION_SUITE_MODULES } from '@fairylights-studio/navigation-suite';
import { MatIconModule } from '@angular/material/icon';
import { MatButtonModule } from '@angular/material/button';`}
/>

### 2. TypeScript — component class

<Source
  language="ts"
  code={`import { Component } from '@angular/core';
import { MAT_NAVIGATION_SUITE_MODULES, MatNavigationSuiteScaffoldState } from '@fairylights-studio/navigation-suite';
import { MatIconModule } from '@angular/material/icon';
import { MatButtonModule } from '@angular/material/button';

@Component({
selector: 'app-my-page',
standalone: true,
imports: [MAT_NAVIGATION_SUITE_MODULES, MatIconModule, MatButtonModule],
templateUrl: './my-page.component.html',
styleUrl: './my-page.component.scss',
})
export class MyPageComponent {
selectedIndex = 0;
scaffoldState = new MatNavigationSuiteScaffoldState();
}`}
/>

### 3. HTML — template

<Source
  language="html"
  code={`<mat-navigation-suite-scaffold
  [navSuiteType]="null"
  [state]="scaffoldState"
  [primaryActionAlignment]="'end'"
>
  <button
    *matNavigationSuitePrimaryAction
    mat-fab
    aria-label="Create document"
  >
    <mat-icon aria-hidden="true">edit</mat-icon>
  </button>

<mat-navigation-suite
[alwaysShowItemLabel]="true"
ariaLabel="Workspace"

>

    <mat-navigation-suite-item
      [selected]="selectedIndex === 0"
      [icon]="'dashboard'"
      [label]="'Dashboard'"
      (click)="selectedIndex = 0"
    />
    <mat-navigation-suite-item
      [selected]="selectedIndex === 1"
      [icon]="'task_alt'"
      [label]="'Tasks'"
      (click)="selectedIndex = 1"
    />
    <mat-navigation-suite-item
      [selected]="selectedIndex === 2"
      [icon]="'analytics'"
      [label]="'Reports'"
      (click)="selectedIndex = 2"
    />
    <mat-navigation-suite-item
      [selected]="selectedIndex === 3"
      [icon]="'admin_panel_settings'"
      [label]="'Admin'"
      (click)="selectedIndex = 3"
    />

  </mat-navigation-suite>

  <section style="padding: 32px">
    <h2>Workspace</h2>
    <p>The scaffold switches between rail and bar layouts.</p>
  </section>
</mat-navigation-suite-scaffold>`}
/>
